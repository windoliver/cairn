```rust
/// Severity level for a lint finding.
#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash)]
pub enum Severity {
    /// Human-review informational finding.
    Info,
    /// Potential correctness issue.
    Warning,
    /// Hard error.
    Error,
}

/// Lint finding category.
#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash)]
pub enum LintKind {
    /// Multiple live edges share the same bitemporal identity triple.
    ContradictoryEdge,
    /// A live edge is marked ambiguous and needs human review.
    AmbiguousEdge,
}

/// Structured lint finding used internally before mapping to generated wire types.
#[derive(Debug, Clone, PartialEq, Eq)]
pub struct LintFinding {
    /// Finding kind.
    pub kind: LintKind,
    /// Severity level.
    pub severity: Severity,
    /// Affected entity or edge ids.
    pub entities: Vec<String>,
    /// Human-readable message.
    pub message: String,
    /// Optional remediation hint.
    pub suggestion: Option<String>,
}

/// Confidence source attached to an entity edge.
#[derive(Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord, Hash)]
pub enum EdgeConfidence {
    /// Human review required.
    Ambiguous,
    /// Inferred by an extractor.
    Inferred,
    /// Directly extracted from source evidence.
    Extracted,
}

impl EdgeConfidence {
    /// Parse the SQLite wire value.
    ///
    /// # Errors
    /// Returns the original string when it is not one of the supported values.
    pub fn parse(value: &str) -> Result<Self, String> {
        match value {
            "AMBIGUOUS" => Ok(Self::Ambiguous),
            "INFERRED" => Ok(Self::Inferred),
            "EXTRACTED" => Ok(Self::Extracted),
            other => Err(other.to_owned()),
        }
    }

    /// SQLite wire value.
    #[must_use]
    pub fn as_str(self) -> &'static str {
        match self {
            Self::Ambiguous => "AMBIGUOUS",
            Self::Inferred => "INFERRED",
            Self::Extracted => "EXTRACTED",
        }
    }
}

/// Candidate live edge in a contradiction group.
#[derive(Debug, Clone, PartialEq)]
pub struct EdgeCandidate {
    /// Edge id.
    pub id: String,
    /// Discrete confidence source.
    pub confidence: EdgeConfidence,
    /// Numeric confidence score.
    pub confidence_score: f64,
}

/// Choose the single edge to keep from a live contradiction group.
#[must_use]
pub fn choose_edge_keeper(edges: &[EdgeCandidate]) -> Option<&EdgeCandidate> {
    edges.iter().max_by(|left, right| {
        left.confidence_score
            .total_cmp(&right.confidence_score)
            .then_with(|| left.confidence.cmp(&right.confidence))
            .then_with(|| right.id.cmp(&left.id))
    })
}
```

```rust
/// SQLite store operation failure.
#[derive(Debug, Error)]
pub enum StoreError {
    /// Underlying SQLite failure.

    SchemaMissing {
        /// Missing table or index name.
        object: &'static str,

    /// Existing edge row contains an unsupported confidence literal.

        /// Edge id.
```

```rust
use rusqlite::Connection;

use crate::StoreError;

const SCHEMA_SQL: &str = r#"
CREATE TABLE IF NOT EXISTS entity_edges (
  id TEXT PRIMARY KEY,
  source_id TEXT NOT NULL,
  target_id TEXT NOT NULL,
  relation TEXT NOT NULL,
  valid_at INTEGER NOT NULL,
  invalid_at INTEGER,

    CHECK (confidence IN ('EXTRACTED', 'INFERRED', 'AMBIGUOUS')),
  confidence_score REAL NOT NULL
    CHECK (confidence_score >= 0.0 AND confidence_score <= 1.0),
  created_at INTEGER NOT NULL
);

CREATE INDEX IF NOT EXISTS entity_edges_live_triple_idx
  ON entity_edges(source_id, target_id, relation)
  WHERE invalid_at IS NULL AND expired_at IS NULL;

CREATE TABLE IF NOT EXISTS wal_ops (
  operation_id TEXT PRIMARY KEY,
  state TEXT NOT NULL,
  kind TEXT NOT NULL,
  reason TEXT NOT NULL,
  envelope TEXT NOT NULL,
  issued_at INTEGER NOT NULL,
  committed_at INTEGER
);

CREATE TABLE IF NOT EXISTS replay_ledger (
  operation_id TEXT PRIMARY KEY,
  reason TEXT NOT NULL,
  committed_at INTEGER NOT NULL
);
"#;

/// Apply the schema needed by bitemporal edge linting.
///
/// # Errors
/// Returns SQLite errors from schema execution.
pub fn migrate(conn: &Connection) -> Result<(), StoreError> {
    conn.execute_batch(SCHEMA_SQL)?;
    Ok(())
}

pub(crate) fn ensure_table(conn: &Connection, table: &'static str) -> Result<(), StoreError> {
    let exists: bool = conn.query_row(
        "SELECT EXISTS (
             SELECT 1 FROM sqlite_master
              WHERE type = 'table' AND name = ?1
         )",
        [table],
        |row| row.get(0),
    )?;
    if exists {
        Ok(())
    } else {
        Err(StoreError::SchemaMissing { object: table })
    }
}
```

```rust
//! Bitemporal entity-edge lint checks.

use std::collections::BTreeMap;

use cairn_core::domain::{
    EdgeCandidate, EdgeConfidence, LintFinding, LintKind, Severity, choose_edge_keeper,
};
use rusqlite::{Connection, params};

use crate::migrations::ensure_table;
use crate::StoreError;

const CONTRADICTION_MESSAGE: &str =
    "Two live edges share (source_id, target_id, relation); one should be invalidated";
const CONTRADICTION_SUGGESTION: &str =
    "Run `cairn lint --fix` to keep the higher-confidence edge";
const AMBIGUOUS_MESSAGE: &str = "Live edge has AMBIGUOUS confidence";
const FIX_REASON: &str = "lint:contradiction_resolution";

/// Lint report returned by the SQLite store.
#[derive(Debug, Clone, PartialEq, Eq)]
pub struct EdgeLintReport {
    /// All lint findings.
    pub findings: Vec<LintFinding>,
    /// Number of contradictory edge pairs.
    pub contradictions: u64,
    /// Number of live ambiguous edges.
    pub ambiguous_edges: u64,
    /// Number of edges invalidated by fix mode.
    pub auto_resolved: u64,
}

#[derive(Debug, Clone)]
struct LiveEdge {
    id: String,
    confidence: EdgeConfidence,
    confidence_score: f64,
}

/// Detect live contradictory and ambiguous entity edges.
///
/// # Errors
/// Returns a schema error when `entity_edges` is absent, or SQLite errors from
/// query execution.
pub fn lint_edges(conn: &Connection) -> Result<EdgeLintReport, StoreError> {
    ensure_table(conn, "entity_edges")?;
    let mut findings = Vec::new();

    let mut contradiction_stmt = conn.prepare(
        "SELECT a.id AS edge_a,
                b.id AS edge_b
           FROM entity_edges a
           JOIN entity_edges b
             ON a.source_id = b.source_id
            AND a.target_id = b.target_id
            AND a.relation = b.relation
            AND a.id < b.id
          WHERE a.invalid_at IS NULL
            AND b.invalid_at IS NULL
            AND a.expired_at IS NULL
            AND b.expired_at IS NULL
          ORDER BY a.id, b.id",
    )?;
    let contradiction_rows = contradiction_stmt.query_map([], |row| {
        Ok((row.get::<_, String>(0)?, row.get::<_, String>(1)?))
    })?;
    for row in contradiction_rows {
        let (edge_a, edge_b) = row?;
        findings.push(LintFinding {
            kind: LintKind::ContradictoryEdge,
            severity: Severity::Warning,
            entities: vec![edge_a, edge_b],
            message: CONTRADICTION_MESSAGE.to_owned(),
            suggestion: Some(CONTRADICTION_SUGGESTION.to_owned()),
        });
    }
    let contradictions = findings.len() as u64;

    let mut ambiguous_stmt = conn.prepare(
        "SELECT id
           FROM entity_edges
          WHERE invalid_at IS NULL
            AND expired_at IS NULL
            AND confidence = 'AMBIGUOUS'
          ORDER BY id",
    )?;
    let ambiguous_rows = ambiguous_stmt.query_map([], |row| row.get::<_, String>(0))?;
    let mut ambiguous_edges = 0_u64;
    for row in ambiguous_rows {
        let edge_id = row?;
        ambiguous_edges += 1;
        findings.push(LintFinding {
            kind: LintKind::AmbiguousEdge,
            severity: Severity::Info,
            entities: vec![edge_id],
            message: AMBIGUOUS_MESSAGE.to_owned(),
            suggestion: None,
        });
    }

    Ok(EdgeLintReport {
        findings,
        contradictions,
        ambiguous_edges,
        auto_resolved: 0,
    })
}

/// Resolve live edge contradictions by invalidating non-keeper edges.
///
/// # Errors
/// Returns schema, confidence parsing, or SQLite errors. The transaction rolls
/// back on every error.
pub fn resolve_edge_contradictions(
    conn: &mut Connection,
    now: i64,
    operation_id: &str,
) -> Result<EdgeLintReport, StoreError> {
    ensure_table(conn, "entity_edges")?;
    ensure_table(conn, "wal_ops")?;
    ensure_table(conn, "replay_ledger")?;

    let tx = conn.transaction()?;
    let groups = live_edge_groups(&tx)?;
    let mut losers = Vec::new();

    for edges in groups.values().filter(|edges| edges.len() > 1) {
        let candidates: Vec<EdgeCandidate> = edges
            .iter()
            .map(|edge| EdgeCandidate {
                id: edge.id.clone(),
                confidence: edge.confidence,
                confidence_score: edge.confidence_score,
            })
            .collect();
        if let Some(keeper) = choose_edge_keeper(&candidates) {
            losers.extend(
                edges
                    .iter()
                    .filter(|edge| edge.id != keeper.id)
                    .map(|edge| edge.id.clone()),
            );
        }
    }

    if losers.is_empty() {
        let mut report = lint_edges(&tx)?;
        report.auto_resolved = 0;
        tx.commit()?;
        return Ok(report);
    }

    tx.execute(
        "INSERT INTO wal_ops (
            operation_id, state, kind, reason, envelope, issued_at, committed_at
         ) VALUES (?1, 'COMMITTED', 'lint_fix', ?2, ?3, ?4, ?4)",
        params![
            operation_id,
            FIX_REASON,
            format!(r#"{{"operation_id":"{operation_id}","reason":"{FIX_REASON}"}}"#),
            now
        ],
    )?;
    tx.execute(
        "INSERT INTO replay_ledger (operation_id, reason, committed_at)
         VALUES (?1, ?2, ?3)",
        params![operation_id, FIX_REASON, now],
    )?;

    for loser in &losers {
        tx.execute(
            "UPDATE entity_edges SET invalid_at = ?1 WHERE id = ?2",
            params![now, loser],
        )?;
    }

    let mut report = lint_edges(&tx)?;
    report.auto_resolved = losers.len() as u64;
    tx.commit()?;
    Ok(report)
}

fn live_edge_groups(
    conn: &Connection,
) -> Result<BTreeMap<(String, String, String), Vec<LiveEdge>>, StoreError> {
    let mut stmt = conn.prepare(
        "SELECT id, source_id, target_id, relation, confidence, confidence_score
           FROM entity_edges
          WHERE invalid_at IS NULL
            AND expired_at IS NULL
          ORDER BY source_id, target_id, relation, id",
    )?;
    let rows = stmt.query_map([], |row| {
        Ok((
            row.get::<_, String>(0)?,
            row.get::<_, String>(1)?,
            row.get::<_, String>(2)?,
            row.get::<_, String>(3)?,
            row.get::<_, String>(4)?,
            row.get::<_, f64>(5)?,
        ))
    })?;

    let mut groups: BTreeMap<(String, String, String), Vec<LiveEdge>> = BTreeMap::new();
    for row in rows {
        let (id, source_id, target_id, relation, confidence_raw, confidence_score) = row?;
        let confidence = EdgeConfidence::parse(&confidence_raw).map_err(|value| {
            StoreError::InvalidConfidence {
                edge_id: id.clone(),
                value,
            }
        })?;
        groups
            .entry((source_id.clone(), target_id.clone(), relation.clone()))
            .or_default()
            .push(LiveEdge {
                id,
                confidence,
                confidence_score,
            });
    }
    Ok(groups)
}
```

```rust
//! `cairn lint` handler.

use std::path::Path;
use std::process::ExitCode;

use cairn_core::domain::{LintFinding, LintKind, Severity};
use cairn_core::generated::envelope::{
    Response, ResponseData, ResponsePolicyTrace, ResponseStatus, ResponseVerb,
};
use cairn_core::generated::verbs::lint::{
    LintData, LintDataFindings, LintDataFindingsKind, LintDataFindingsSeverity,
    LintDataSummary,
};
use cairn_store_sqlite::{StoreError, lint_edges, migrate, resolve_edge_contradictions};
use clap::ArgMatches;
use rusqlite::{Connection, OpenFlags};

use super::envelope::{emit_json, human_error, new_operation_id};

/// Run `cairn lint`.
#[must_use]
pub fn run(sub: &ArgMatches) -> ExitCode {
    let json = sub.get_flag("json");
    let fix = sub.get_flag("fix");
    let operation_id = new_operation_id();

    let result = run_lint(fix, &operation_id.0);
    match result {
        Ok(data) => {
            let has_warnings = data
                .findings
                .iter()
                .any(|finding| {
                    matches!(
                        finding.severity,
                        Some(LintDataFindingsSeverity::Warning | LintDataFindingsSeverity::Error)
                    )
                });
            let response = committed_response(operation_id, data);
            if json {
                emit_json(&response);
            } else {
                print_human(&response);
            }
            if has_warnings && !fix {
                ExitCode::from(1)
            } else {
                ExitCode::SUCCESS
            }
        }
        Err(err) => {
            let message = err.to_string();
            let response = aborted_response(operation_id, &message);
            if json {
                emit_json(&response);
            } else {
                human_error("lint", "Internal", &message, &response.operation_id);
            }
            ExitCode::from(1)
        }
    }
}

fn run_lint(fix: bool, operation_id: &str) -> Result<LintData, StoreError> {
    let db_path = db_path(std::env::current_dir().unwrap_or_else(|_| ".".into()).as_path());
    let report = if fix {
        let mut conn = Connection::open(db_path)?;
        migrate(&conn)?;
        resolve_edge_contradictions(&mut conn, unix_now_seconds(), operation_id)?
    } else {
        let conn = Connection::open_with_flags(db_path, OpenFlags::SQLITE_OPEN_READ_ONLY)?;
        lint_edges(&conn)?
    };
    Ok(map_report(report))
}

fn db_path(vault_root: &Path) -> std::path::PathBuf {
    vault_root.join(".cairn/cairn.db")
}

fn unix_now_seconds() -> i64 {
    std::time::SystemTime::now()
        .duration_since(std::time::UNIX_EPOCH)
        .map_or(0, |duration| duration.as_secs() as i64)
}

fn committed_response(operation_id: cairn_core::generated::common::Ulid, data: LintData) -> Response {
    Response {
        contract: "cairn.mcp.v1".to_owned(),
        data: Some(ResponseData::Lint(data)),
        error: None,
        operation_id,
        policy_trace: Vec::<ResponsePolicyTrace>::new(),
        status: ResponseStatus::Committed,
        target: None,
        verb: ResponseVerb::Lint,
    }
}

fn aborted_response(operation_id: cairn_core::generated::common::Ulid, message: &str) -> Response {
    Response {
        contract: "cairn.mcp.v1".to_owned(),
        data: None,
        error: Some(serde_json::json!({
            "code": "Internal",
            "message": message,
        })),
        operation_id,
        policy_trace: Vec::<ResponsePolicyTrace>::new(),
        status: ResponseStatus::Aborted,
        target: None,
        verb: ResponseVerb::Lint,
    }
}

fn print_human(response: &Response) {
    if let Some(ResponseData::Lint(data)) = &response.data {
        println!(
            "cairn lint: {} findings, {} auto-resolved",
            data.summary.total,
            data.summary.auto_resolved.unwrap_or(0)
        );
        for finding in &data.findings {
            println!("- {:?}: {}", finding.kind, finding.message);
        }
    }
}

fn map_report(report: cairn_store_sqlite::EdgeLintReport) -> LintData {
    let total = report.findings.len() as u64;
    LintData {
        findings: report.findings.into_iter().map(map_finding).collect(),
        report_path: None,
        summary: LintDataSummary {
            ambiguous_edges: Some(report.ambiguous_edges),
            auto_resolved: Some(report.auto_resolved),
            contradictions: Some(report.contradictions),
            orphans: None,
            stale: None,
            total,
        },
    }
}

fn map_finding(finding: LintFinding) -> LintDataFindings {
    LintDataFindings {
        entities: if finding.entities.is_empty() {
            None
        } else {
            Some(finding.entities)
        },
        kind: match finding.kind {
            LintKind::ContradictoryEdge => LintDataFindingsKind::ContradictoryEdge,
            LintKind::AmbiguousEdge => LintDataFindingsKind::AmbiguousEdge,
        },
        message: finding.message,
        record_id: None,
        severity: Some(match finding.severity {
            Severity::Info => LintDataFindingsSeverity::Info,
            Severity::Warning => LintDataFindingsSeverity::Warning,
            Severity::Error => LintDataFindingsSeverity::Error,
        }),
        suggestion: finding.suggestion,
    }
}
```
