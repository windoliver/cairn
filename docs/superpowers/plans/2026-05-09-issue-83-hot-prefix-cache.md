# Issue #83 — Hot-prefix cache + lint + metrics — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

```rust
//! Hot-prefix source classification — see issue #83 / brief §7.
//!
//! The hot-memory prefix is invalidated by a watermark counter per
//! source class. Every record-mutating write classifies the touched
//! record(s) and bumps the matching counter(s) inside the same SQLite
//! transaction. Cache rows snapshot all six watermarks at assembly
//! time; a later cache read is a hit only when every field still
//! matches the live counter.

use serde::{Deserialize, Serialize};

/// One bucket of the hot-prefix invalidation tree.
#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash, Serialize, Deserialize)]
#[serde(rename_all = "snake_case")]
#[non_exhaustive]
pub enum SourceClass {
    /// `user`, `feedback`, `entity`, `strategy` records that feed
    /// `AutoUserProfile` (§7.1).
    ProfileEvidence,
    /// Records with `pinned = true`.
    Pinned,
    /// Vault-root `purpose.md` and `index.md`.
    PurposeIndex,
    /// `_summary.md` files at any folder depth.
    Summaries,
    /// `playbook` records.
    Playbooks,
    /// `.cairn/config.yaml` `hot_memory.*` keys.
    Policy,
}

impl SourceClass {
    /// All variants in stable iteration order.
    pub const ALL: [Self; 6] = [
        Self::ProfileEvidence,
        Self::Pinned,
        Self::PurposeIndex,
        Self::Summaries,
        Self::Playbooks,
        Self::Policy,
    ];

    /// Snake-case key used in SQLite + JSONL serialisations.
    #[must_use]
    pub fn as_db_str(self) -> &'static str {
        match self {
            Self::ProfileEvidence => "profile_evidence",
            Self::Pinned => "pinned",
            Self::PurposeIndex => "purpose_index",
            Self::Summaries => "summaries",
            Self::Playbooks => "playbooks",
            Self::Policy => "policy",
        }
    }

    /// Inverse of [`Self::as_db_str`]; `None` on unknown input.
    #[must_use]
    pub fn parse(raw: &str) -> Option<Self> {
        match raw {
            "profile_evidence" => Some(Self::ProfileEvidence),
            "pinned" => Some(Self::Pinned),
            "purpose_index" => Some(Self::PurposeIndex),
            "summaries" => Some(Self::Summaries),
            "playbooks" => Some(Self::Playbooks),
            "policy" => Some(Self::Policy),
            _ => None,
        }
    }
}

/// Snapshot of every source-class watermark.
#[derive(Debug, Default, Clone, Copy, PartialEq, Eq, Serialize, Deserialize)]
pub struct SourceWatermarks {
    pub profile_evidence: u64,
    pub pinned: u64,
    pub purpose_index: u64,
    pub summaries: u64,
    pub playbooks: u64,
    pub policy: u64,
}

impl SourceWatermarks {
    /// Field-wise equality.
    #[must_use]
    pub fn matches(&self, other: &Self) -> bool {
        self == other
    }

    /// Increment the counter for `class`.
    pub fn bump(&mut self, class: SourceClass) {
        match class {
            SourceClass::ProfileEvidence => self.profile_evidence += 1,
            SourceClass::Pinned => self.pinned += 1,
            SourceClass::PurposeIndex => self.purpose_index += 1,
            SourceClass::Summaries => self.summaries += 1,
            SourceClass::Playbooks => self.playbooks += 1,
            SourceClass::Policy => self.policy += 1,
        }
    }
}

#[cfg(test)]
mod tests {
    // (the three tests written in Task 3)
    use super::*;

    #[test]
    fn watermarks_match_is_reflexive() {
        let w = SourceWatermarks::default();
        assert!(w.matches(&w));
    }

    #[test]
    fn watermarks_match_breaks_when_any_field_diverges() {
        let base = SourceWatermarks::default();
        for class in SourceClass::ALL {
            let mut other = base;
            other.bump(class);
            assert!(!base.matches(&other), "class {class:?} did not invalidate match");
        }
    }

    #[test]
    fn source_class_all_returns_six_classes() {
        assert_eq!(SourceClass::ALL.len(), 6);
    }

    #[test]
    fn source_class_round_trips_through_db_str() {
        for class in SourceClass::ALL {
            assert_eq!(SourceClass::parse(class.as_db_str()), Some(class));
        }
    }
}
```

```rust
use crate::domain::MemoryRecord;
use crate::domain::taxonomy::MemoryKind;
use smallvec::SmallVec;

/// Classify a record into the source classes whose watermarks must be
/// bumped when this record is upserted, updated, or tombstoned.
///
/// Pure function — every store-side write path calls this; lint and
/// tests reuse the same mapping.
#[must_use]
pub fn classify_record(r: &MemoryRecord) -> SmallVec<[SourceClass; 2]> {
    let mut out = SmallVec::new();
    let pinned = is_pinned(r);
    match r.kind {
        MemoryKind::User
        | MemoryKind::Feedback
        | MemoryKind::Entity
        | MemoryKind::Strategy => {
            out.push(SourceClass::ProfileEvidence);
            if pinned {
                out.push(SourceClass::Pinned);
            }
        }
        MemoryKind::Project if pinned => out.push(SourceClass::Pinned),
        MemoryKind::Playbook => out.push(SourceClass::Playbooks),
        _ => {}
    }
    out
}

/// Mirror of `cairn-cli`'s `is_pinned_record` — kept private to this
/// module because the canonical predicate lives in §7 (the `pinned`
/// field on the record), not in the CLI.
fn is_pinned(r: &MemoryRecord) -> bool {
    r.pinned
}
```

```rust
#[non_exhaustive]
pub enum CacheError {

    #[error("hot-prefix cache backend: {0}")]

    /// Look up a cached entry. `Ok(None)` for cache miss.

        agent: &Identity,
        recipe_hash: &str,

    async fn put(
        &self,
        agent: &Identity,
        recipe_hash: &str,
        entry: &CachedPrefix,
    ) -> Result<(), CacheError>;

    /// transaction by the store; out-of-tx callers exist only for
    /// non-record-bound triggers (config write, summary write).
    async fn bump(&self, classes: &[SourceClass]) -> Result<SourceWatermarks, CacheError>;
}
```

```rust
use crate::config::HotMemoryConfig;

    Assemble(#[from] AssembleHotError),
}

#[must_use]
pub fn recipe_hash_canonical(recipe: &[crate::config::HotMemoryRecipeStep]) -> String {

fn hex_lower(bytes: &[u8]) -> String {
    let mut out = String::with_capacity(bytes.len() * 2);
    for b in bytes {
        use std::fmt::Write as _;
        let _ = write!(out, "{b:02x}");
    }
    out
```

```rust
pub async fn cached_assemble(
    config: &HotMemoryConfig,
    agent: &Identity,
    vault_id: &str,
    cache: &dyn HotPrefixCache,
    metrics: &dyn MetricsSink,
    load_bodies: impl FnOnce() -> Result<Vec<String>, AssembleHotError>,
) -> Result<AssembleHotData, CachedAssembleError> {
    let recipe_hash = recipe_hash_canonical(&config.recipe);
    let started = Instant::now();

    let wm_now: SourceWatermarks = match cache.current_watermarks().await {
        Ok(wm) => wm,
        Err(e) => {
            tracing::warn!(error = %e, "hot-prefix cache: watermark read failed; bypassing cache");
            return assemble_and_emit(
                config, agent, vault_id, &recipe_hash, metrics, started,
                false, SourceWatermarks::default(), load_bodies,
            ).await;
        }
    };

    let cache_get = cache.get(agent, &recipe_hash).await;
    if let Ok(Some(entry)) = cache_get {
        if entry.watermarks.matches(&wm_now) {
            let latency_ms = elapsed_ms(started);
            emit_hit(metrics, vault_id, agent, &recipe_hash, &entry, latency_ms, wm_now,
                     u64::from(config.max_bytes)).await;
            return Ok(into_assemble_hot_data(entry));
        }
    } else if let Err(e) = cache_get {
        tracing::warn!(error = %e, "hot-prefix cache: get failed; bypassing cache for this call");
    }

    let bodies = load_bodies()?;
    let data = assemble_hot_from_bodies(config, bodies, None)?;
    let latency_ms = elapsed_ms(started);

    let entry = CachedPrefix {
        prefix: data.prefix.as_bytes().to_vec(),
        segments: data.segments.clone().unwrap_or_default(),
        bytes: data.bytes,
        watermarks: wm_now,
        assembled_at_ms: now_ms(),
        assembly_latency_ms: latency_ms,
    };
    if let Err(e) = cache.put(agent, &recipe_hash, &entry).await {
        tracing::warn!(error = %e, "hot-prefix cache: put failed");
    }

    emit_miss(metrics, vault_id, agent, &recipe_hash, &entry, latency_ms, wm_now,
              u64::from(config.max_bytes)).await;

    Ok(data)
}

async fn assemble_and_emit(
    config: &HotMemoryConfig,
    agent: &Identity,
    vault_id: &str,
    recipe_hash: &str,
    metrics: &dyn MetricsSink,
    started: Instant,
    cache_hit: bool,
    wm: SourceWatermarks,
    load_bodies: impl FnOnce() -> Result<Vec<String>, AssembleHotError>,
) -> Result<AssembleHotData, CachedAssembleError> {
    let bodies = load_bodies()?;
    let data = assemble_hot_from_bodies(config, bodies, None)?;
    let latency_ms = elapsed_ms(started);
    let event = MetricEvent::HotPrefixAssembled {
        ts_ms: now_ms(),
        vault_id: vault_id.to_owned(),
        agent_id: agent.to_string(),
        recipe_hash: recipe_hash.to_owned(),
        latency_ms,
        bytes: data.bytes,
        budget_bytes: u64::from(config.max_bytes),
        budget_used_ratio: budget_ratio(data.bytes, u64::from(config.max_bytes)),
        cache_hit,
        watermarks: wm,
    };
    if let Err(e) = metrics.emit(event).await {
        tracing::warn!(error = %e, "metrics sink emit failed");
    }
    Ok(data)
}

fn elapsed_ms(start: Instant) -> u64 {
    u64::try_from(start.elapsed().as_millis()).unwrap_or(u64::MAX)

fn now_ms() -> i64 {
    use std::time::SystemTime;
    SystemTime::now()

        .map(|d| i64::try_from(d.as_millis()).unwrap_or(i64::MAX))
        .unwrap_or(0)
}

fn budget_ratio(used: u64, budget: u64) -> f64 {
    if budget == 0 { 0.0 } else { used as f64 / budget as f64 }
```

```rust
//! `HotPrefixCache` implementation backed by the same SQLite database
//! as the records store. See issue #83 / spec §5.

use std::path::Path;
use std::sync::Arc;

use cairn_core::contract::hot_prefix_cache::{CacheError, CachedPrefix, HotPrefixCache};
use cairn_core::domain::Identity;
use cairn_core::domain::hot_prefix::{SourceClass, SourceWatermarks};

use crate::open::AsyncConn;

pub struct SqliteHotPrefixCache {
    conn: Arc<AsyncConn>,
}

impl SqliteHotPrefixCache {
    /// Open or attach to the cairn vault at `vault_root`. Returns an
    /// instance bound to `<vault_root>/.cairn/cairn.db`.
    pub async fn open(vault_root: &Path) -> Result<Self, CacheError> {
        let db = vault_root.join(".cairn").join("cairn.db");
        let conn = crate::open::open_async(&db)
            .await
            .map_err(|e| CacheError::Backend(Box::new(e)))?;
        Ok(Self { conn })
    }

    /// Construct from an existing `AsyncConn` — used by the CLI when
    /// the store is already opened.
    #[must_use]
    pub fn from_conn(conn: Arc<AsyncConn>) -> Self { Self { conn } }
}

#[async_trait::async_trait]
impl HotPrefixCache for SqliteHotPrefixCache {
    async fn current_watermarks(&self) -> Result<SourceWatermarks, CacheError> {
        let conn = self.conn.clone();
        conn.call(|c| {
            let mut stmt = c
                .prepare("SELECT class, watermark FROM hot_source_watermarks")
                .map_err(rusqlite_to_cache)?;
            let rows = stmt
                .query_map([], |r| {
                    let class: String = r.get(0)?;
                    let watermark: i64 = r.get(1)?;
                    Ok((class, watermark))
                })
                .map_err(rusqlite_to_cache)?;
            let mut wm = SourceWatermarks::default();
            let mut seen = [false; 6];
            for row in rows {
                let (class, watermark) = row.map_err(rusqlite_to_cache)?;
                let Some(c) = SourceClass::parse(&class) else {
                    continue; // unknown class — forward-compat
                };
                wm.bump_to(c, u64::try_from(watermark).unwrap_or(0));
                seen[c as usize] = true;
            }
            for (i, was_seen) in seen.iter().enumerate() {
                if !was_seen {
                    return Err(CacheError::WatermarkSchemaMismatch {
                        class: SourceClass::ALL[i],
                    });
                }
            }
            Ok(wm)
        })
        .await
        .map_err(|e| CacheError::Backend(Box::new(e)))?
    }

    async fn get(
        &self,
        agent: &Identity,
        recipe_hash: &str,
    ) -> Result<Option<CachedPrefix>, CacheError> {
        let agent_id = agent.to_string();
        let recipe_hash = recipe_hash.to_owned();
        let conn = self.conn.clone();
        conn.call(move |c| {
            let mut stmt = c
                .prepare(
                    "SELECT prefix, segments_json, bytes, watermarks_json, \
                     assembled_at_ms, assembly_latency_ms \
                     FROM hot_prefix_cache WHERE agent_id = ?1 AND recipe_hash = ?2",
                )
                .map_err(rusqlite_to_cache)?;
            let row = stmt
                .query_row((&agent_id, &recipe_hash), |r| {
                    let prefix: Vec<u8> = r.get(0)?;
                    let segments_json: String = r.get(1)?;
                    let bytes: i64 = r.get(2)?;
                    let watermarks_json: String = r.get(3)?;
                    let assembled_at_ms: i64 = r.get(4)?;
                    let assembly_latency_ms: i64 = r.get(5)?;
                    Ok((prefix, segments_json, bytes, watermarks_json, assembled_at_ms, assembly_latency_ms))
                })
                .map(Some)
                .or_else(|e| match e {
                    rusqlite::Error::QueryReturnedNoRows => Ok(None),
                    other => Err(rusqlite_to_cache(other)),
                })?;

            let Some((prefix, seg_json, bytes, wm_json, at, lat)) = row else { return Ok(None) };
            let segments: Vec<cairn_core::generated::verbs::assemble_hot::Segment> =
                serde_json::from_str(&seg_json)
                    .map_err(|e| CacheError::Corrupt { reason: format!("segments: {e}") })?;
            let watermarks: SourceWatermarks = serde_json::from_str(&wm_json)
                .map_err(|e| CacheError::Corrupt { reason: format!("watermarks: {e}") })?;
            Ok(Some(CachedPrefix {
                prefix,
                segments,
                bytes: u64::try_from(bytes).unwrap_or(0),
                watermarks,
                assembled_at_ms: at,
                assembly_latency_ms: u64::try_from(lat).unwrap_or(0),
            }))
        })
        .await
        .map_err(|e| CacheError::Backend(Box::new(e)))?
    }

    async fn put(
        &self,
        agent: &Identity,
        recipe_hash: &str,
        entry: &CachedPrefix,
    ) -> Result<(), CacheError> {
        let agent_id = agent.to_string();
        let recipe_hash = recipe_hash.to_owned();
        let segments_json = serde_json::to_string(&entry.segments)
            .map_err(|e| CacheError::Corrupt { reason: format!("seg ser: {e}") })?;
        let watermarks_json = serde_json::to_string(&entry.watermarks)
            .map_err(|e| CacheError::Corrupt { reason: format!("wm ser: {e}") })?;
        let prefix = entry.prefix.clone();
        let bytes = i64::try_from(entry.bytes).unwrap_or(i64::MAX);
        let assembled_at_ms = entry.assembled_at_ms;
        let assembly_latency_ms = i64::try_from(entry.assembly_latency_ms).unwrap_or(i64::MAX);
        let conn = self.conn.clone();
        conn.call(move |c| {
            c.execute(
                "INSERT OR REPLACE INTO hot_prefix_cache \
                 (agent_id, recipe_hash, prefix, segments_json, bytes, \
                  watermarks_json, assembled_at_ms, assembly_latency_ms) \
                 VALUES (?1, ?2, ?3, ?4, ?5, ?6, ?7, ?8)",
                rusqlite::params![
                    agent_id, recipe_hash, prefix, segments_json, bytes,
                    watermarks_json, assembled_at_ms, assembly_latency_ms,
                ],
            )
            .map(|_| ())
            .map_err(rusqlite_to_cache)
        })
        .await
        .map_err(|e| CacheError::Backend(Box::new(e)))?
    }

    async fn bump(&self, classes: &[SourceClass]) -> Result<SourceWatermarks, CacheError> {
        let now_ms = now_ms();
        let class_strs: Vec<&str> = classes.iter().map(|c| c.as_db_str()).collect();
        let placeholders = (0..class_strs.len())
            .map(|i| format!("?{}", i + 2))
            .collect::<Vec<_>>()
            .join(",");
        let owned_strs: Vec<String> = class_strs.iter().map(|s| (*s).to_owned()).collect();
        let conn = self.conn.clone();
        conn.call(move |c| {
            if owned_strs.is_empty() { return Ok(()); }
            let sql = format!(
                "UPDATE hot_source_watermarks SET watermark = watermark + 1, \
                 updated_at_ms = ?1 WHERE class IN ({placeholders})"
            );
            let mut stmt = c.prepare(&sql).map_err(rusqlite_to_cache)?;
            let mut params: Vec<&dyn rusqlite::ToSql> = vec![&now_ms];
            for s in &owned_strs { params.push(s); }
            stmt.execute(params.as_slice()).map_err(rusqlite_to_cache)?;
            Ok(())
        })
        .await
        .map_err(|e| CacheError::Backend(Box::new(e)))??;
        self.current_watermarks().await
    }
}

fn now_ms() -> i64 {
    use std::time::SystemTime;
    SystemTime::now()
        .duration_since(SystemTime::UNIX_EPOCH)
        .map(|d| i64::try_from(d.as_millis()).unwrap_or(i64::MAX))
        .unwrap_or(0)
}

fn rusqlite_to_cache(e: rusqlite::Error) -> CacheError {
    CacheError::Backend(Box::new(e))
}

// `bump_to` helper for `current_watermarks` reconstruction.
trait BumpTo { fn bump_to(&mut self, c: SourceClass, v: u64); }
impl BumpTo for SourceWatermarks {
    fn bump_to(&mut self, c: SourceClass, v: u64) {
        match c {
            SourceClass::ProfileEvidence => self.profile_evidence = v,
            SourceClass::Pinned => self.pinned = v,
            SourceClass::PurposeIndex => self.purpose_index = v,
            SourceClass::Summaries => self.summaries = v,
            SourceClass::Playbooks => self.playbooks = v,
            SourceClass::Policy => self.policy = v,
        }
    }
}
```

```rust
use std::sync::Arc;

use cairn_core::contract::metrics::{MetricsError, MetricsSink};
use cairn_core::domain::metrics::MetricEvent;
use tokio::io::AsyncWriteExt;
use tokio::sync::Mutex;

pub struct JsonlMetricsSink {
    path: PathBuf,
    file: Arc<Mutex<tokio::fs::File>>,
}

impl JsonlMetricsSink {
    /// Open or create `<vault_root>/.cairn/metrics.jsonl`. Append-only.
    pub async fn open(vault_root: &Path) -> Result<Self, MetricsError> {
        let path = vault_root.join(".cairn").join("metrics.jsonl");
        if let Some(parent) = path.parent() {
            tokio::fs::create_dir_all(parent).await?;
        }
        let file = tokio::fs::OpenOptions::new()
            .create(true)
            .append(true)
            .open(&path)
            .await?;
        Ok(Self { path, file: Arc::new(Mutex::new(file)) })
    }
    pub fn path(&self) -> &Path { &self.path }
}

#[async_trait::async_trait]
impl MetricsSink for JsonlMetricsSink {
    async fn emit(&self, event: MetricEvent) -> Result<(), MetricsError> {
        let mut line = serde_json::to_vec(&event)?;
        line.push(b'\n');
        let mut guard = self.file.lock().await;
        guard.write_all(&line).await?;
        guard.flush().await?;
        Ok(())
    }
}
```

```rust
/// Pure: scans `profile_body` for `[01...]` ULID-bracketed evidence
/// citations; emits one `StaleProfileLine` Warning per citation
/// whose id does not appear in `active_record_ids`. Body parsing is
/// intentionally simple — the synthesiser controls the citation
/// format (#82's UserProfileSynthesizer).
#[must_use]
pub fn stale_profile_lines(profile_body: &str, active_record_ids: &[String]) -> Vec<Finding> {
    let active: std::collections::HashSet<&str> =
        active_record_ids.iter().map(String::as_str).collect();
    let mut out = Vec::new();
    let mut cursor = profile_body;
    while let Some(open) = cursor.find('[') {
        let after_open = &cursor[open + 1..];
        let Some(close) = after_open.find(']') else { break };
        let candidate = &after_open[..close];
        if is_ulid_like(candidate) && !active.contains(candidate) {
            let mut f = finding(
                Kind::StaleProfileLine,
                Severity::Warning,
                format!(
                    "AutoUserProfile cites forgotten or expired evidence record {candidate}"
                ),
            );
            f.target = None;
            out.push(f);
        }
        cursor = &after_open[close + 1..];
    }
    out
}

fn is_ulid_like(s: &str) -> bool {
    s.len() == 26
        && s.bytes().all(|b| b.is_ascii_alphanumeric())
}
```

```rust
#[must_use]
pub fn run(inputs: &LintInputs<'_>) -> Vec<Finding> {
    let mut findings = Vec::new();

    // 1. Real walker — only when the dispatch layer wired a loader.
    if let Some(loader) = inputs.hot_body_loader {
        let recipe: Vec<crate::generated::verbs::assemble_hot::HotRecipeStep> = inputs
            .config
            .vault
            .hot_memory
            .recipe
            .iter()
            .copied()
            .map(crate::generated::verbs::assemble_hot::HotRecipeStep::from)
            .collect();
        let bodies: Result<Vec<String>, String> = recipe.iter().copied().map(loader).collect();
        match bodies {
            Ok(bodies) => {
                match crate::verbs::assemble_hot::assembler::assemble_hot_from_bodies(
                    &inputs.config.vault.hot_memory,
                    bodies,
                    None,
                ) {
                    Ok(data) => {
                        let max_bytes = u64::from(inputs.config.vault.hot_memory.max_bytes);
                        if data.bytes > max_bytes {
                            findings.push(over_budget_finding(data.bytes, max_bytes));
                        }
                    }
                    Err(e) => {
                        findings.push(crate::verbs::lint::finding(
                            Kind::DeferredCheck,
                            Severity::Warning,
                            format!("hot_memory walker failed: {e}"),
                        ));
                    }
                }
            }
            Err(e) => {
                findings.push(crate::verbs::lint::finding(
                    Kind::DeferredCheck,
                    Severity::Warning,
                    format!("hot_memory loader returned error: {e}"),
                ));
            }
        }
    }

    // 2. broken_source_link.
    findings.extend(crate::verbs::lint::checks::hot_memory_walker::broken_source_links(inputs));

    // 3. missing_summary — folders to check come from config.
    let folders: Vec<String> = inputs
        .config
        .vault
        .hot_memory
        .summary_folders
        .clone()
        .unwrap_or_default();
    findings.extend(crate::verbs::lint::checks::hot_memory_walker::missing_summaries(inputs, &folders));

    // 4. stale_profile_line — only when a profile body is present.
    let active_ids: Vec<String> = inputs
        .records
        .iter()
        .map(|r| r.stored.record.id.as_str().to_owned())
        .collect();
    if let Some(body) = profile_body_from_records(inputs.records) {
        findings.extend(
            crate::verbs::lint::checks::hot_memory_walker::stale_profile_lines(&body, &active_ids),
        );
    }

    findings
}

fn over_budget_finding(got: u64, max: u64) -> Finding {
    let mut f = crate::verbs::lint::finding(
        Kind::HotMemoryOverBudget,
        Severity::Error,
        format!("hot prefix {got} bytes exceeds {max} bytes budget"),
    );
    f.target = Some(crate::verbs::lint::target_path(".cairn/config.yaml"));
    f.tracking_issue = Some(83);
    f
}

fn profile_body_from_records(records: &[crate::verbs::lint::LintRecord]) -> Option<String> {
    // The synthesiser writes a record of kind=user with a tag
    // `auto_user_profile` (issue #81's contract). If the tag changes,

        if r.stored.record.tags.iter().any(|t| t == "auto_user_profile") {
            Some(r.stored.record.body.clone())
        } else {
            None
        }
    })
}
```

```rust
//! Smoke test: brief §15 SLO p95 < 50ms cold + < 5ms warm on the fixture vault.

use std::process::Command;

fn cargo_bin() -> std::path::PathBuf { env!("CARGO_BIN_EXE_cairn").into() }

#[test]
fn assemble_hot_p95_meets_slo_on_fixture_vault() {
    let vault = tempfile::tempdir().expect("vault");
    Command::new(cargo_bin())
        .args(["bootstrap", "--vault", vault.path().to_str().unwrap()])
        .output().expect("bootstrap");

    for _ in 0..10 {
        Command::new(cargo_bin())
            .args(["assemble_hot", "--vault", vault.path().to_str().unwrap(), "--json"])
            .output().expect("call");
    }

    let metrics = vault.path().join(".cairn/metrics.jsonl");
    let content = std::fs::read_to_string(&metrics).expect("read");
    let mut latencies: Vec<u64> = content
        .lines()
        .filter_map(|l| serde_json::from_str::<serde_json::Value>(l).ok())
        .filter_map(|v| v["latency_ms"].as_u64())
        .collect();
    assert!(latencies.len() >= 10, "expected >=10 events; got {}", latencies.len());
    latencies.sort_unstable();
    let p95_idx = (latencies.len() as f64 * 0.95).ceil() as usize - 1;
    let p95 = latencies[p95_idx];
    assert!(p95 < 50, "p95 latency {p95} >= 50 ms (brief §15 SLO)");
}
```
